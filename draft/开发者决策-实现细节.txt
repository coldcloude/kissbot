问：rust中，要定义一个使用Arc<String>和DashMap的struct，要和另一个使用String和HashMap的stuct保持一致，能在编译时检查，怎么办？
答：定义trait来描述“字符串类型”和“映射类型”在并发/单线程下的不同实现，然后通过泛型参数统一结构体。

```rust
use std::collections::HashMap;
use std::sync::Arc;
use dashmap::DashMap;
use std::hash::Hash;

// 定义字符串类型的 trait
trait StringKind {
    type Type: Clone + ?Sized;
    fn from_plain(s: String) -> Self::Type;
}

struct SyncString;
impl StringKind for SyncString {
    type Type = Arc<String>;
    fn from_plain(s: String) -> Self::Type {
        Arc::new(s)
    }
}

struct LocalString;
impl StringKind for LocalString {
    type Type = String;
    fn from_plain(s: String) -> Self::Type {
        s
    }
}

// 定义映射类型的 trait
trait MapKind {
    type Map<K, V>: where K: Eq + Hash;
    fn new_map<K: Eq + Hash, V>() -> Self::Map<K, V>;
}

struct SyncMap;
impl MapKind for SyncMap {
    type Map<K, V> = DashMap<K, V>;
    fn new_map<K: Eq + Hash, V>() -> Self::Map<K, V> {
        DashMap::new()
    }
}

struct LocalMap;
impl MapKind for LocalMap {
    type Map<K, V> = HashMap<K, V>;
    fn new_map<K: Eq + Hash, V>() -> Self::Map<K, V> {
        HashMap::new()
    }
}

// 统一的数据结构
struct MyData<S: StringKind, M: MapKind> {
    name: S::Type,
    map: M::Map<String, i32>,
}

// 为常用组合定义类型别名
type SyncData = MyData<SyncString, SyncMap>;
type LocalData = MyData<LocalString, LocalMap>;

fn main() {
    let sync_data = SyncData {
        name: Arc::new("hello".to_string()),
        map: DashMap::new(),
    };
    let local_data = LocalData {
        name: "hello".to_string(),
        map: HashMap::new(),
    };
}
```
